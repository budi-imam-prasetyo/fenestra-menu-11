cmake_minimum_required(VERSION 3.24)
project(fenestra_menu11_native LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_AUTOMOC ON)

include(GNUInstallDirs)

find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2)

# Prefer a system CXX-Qt (distro package or a previous install). Fetching is a
# development convenience only; packagers should provide CxxQt themselves.
find_package(CxxQt 0.10 QUIET)
if(NOT CxxQt_FOUND)
    include(FetchContent)
    FetchContent_Declare(CxxQt
        GIT_REPOSITORY https://github.com/KDAB/cxx-qt-cmake.git
        GIT_TAG 0.10.0
    )
    FetchContent_MakeAvailable(CxxQt)
endif()

set(QML_URI "org.fenestra.menu11.core")
set(QML_OUTPUT_DIR "${CMAKE_BINARY_DIR}/qml/org/fenestra/menu11/core"
    CACHE PATH "Directory where the loadable QML module is assembled")

cxx_qt_import_crate(
    MANIFEST_PATH "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml"
    CRATES fenestra_qml
    LOCKED
    QT_MODULES Qt::Core Qt::Gui Qt::Qml Qt::QuickControls2
)

# For a staticlib crate cxx_qt_import_qml_module exports the generated
# initializer object; the final plugin library must be assembled explicitly.
cxx_qt_import_qml_module(fenestra_qml_module
    URI "${QML_URI}"
    SOURCE_CRATE fenestra_qml
)

add_library(fenestra_qml_plugin MODULE
    $<TARGET_OBJECTS:fenestra_qml_module>
)
target_link_libraries(fenestra_qml_plugin PRIVATE
    fenestra_qml
    Qt6::Core
    Qt6::Gui
    Qt6::Qml
    Qt6::QuickControls2
)

# The plugin object generated by CXX-Qt is only referenced by the QML module
# initializer, so force the linker to retain its qt_plugin_instance export.
if(UNIX AND NOT APPLE)
    target_link_options(fenestra_qml_plugin PRIVATE "-Wl,-u,qt_plugin_instance")
endif()

set_target_properties(fenestra_qml_plugin PROPERTIES
    OUTPUT_NAME "org_fenestra_menu11_core"
    LIBRARY_OUTPUT_DIRECTORY "${QML_OUTPUT_DIR}"
)

# qmldir describes the file-system module used by the widget's relative import.
configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/qml/org/fenestra/menu11/core/qmldir"
    "${QML_OUTPUT_DIR}/qmldir"
    COPYONLY
)

# qmltypes is generated by CXX-Qt; copy it next to the plugin when available.
add_custom_command(TARGET fenestra_qml_plugin POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${CMAKE_BINARY_DIR}/cxxqt/qml_modules/org/fenestra/menu11/core/plugin.qmltypes"
        "${QML_OUTPUT_DIR}/plugin.qmltypes"
    VERBATIM
)

install(TARGETS fenestra_qml_plugin
    LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/qml/org/fenestra/menu11/core"
)
install(FILES
    "${CMAKE_CURRENT_SOURCE_DIR}/qml/org/fenestra/menu11/core/qmldir"
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/qml/org/fenestra/menu11/core"
)
install(FILES
    "${CMAKE_BINARY_DIR}/cxxqt/qml_modules/org/fenestra/menu11/core/plugin.qmltypes"
    DESTINATION "${CMAKE_INSTALL_LIBDIR}/qml/org/fenestra/menu11/core"
    OPTIONAL
)
